% AON node: top row ES-Dur-EF ; middle row center = label ; bottom row LS-Float-LF
\newcommand{\cpmnode}[9]{%
  % #1=name, #2=label, #3=pos, #4=ES, #5=Dur, #6=EF, #7=LS, #8=Float, #9=LF
  \node[draw,very thick,inner sep=0pt,align=center] (#1) at #3 {%
    \begin{tabular}{|p{1.1cm}|p{1.1cm}|p{1.1cm}|}
    \hline
    \centering #4 & \centering #5 & \centering #6 \\ \hline
     & \textbf{#2} &  \\ \hline
    \centering #7 & \centering #8 & \centering #9 \\ \hline
    \end{tabular}
  };
}
